cmake_minimum_required(VERSION 3.20)
project(abmio VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

option(ABMIO_BUILD_TESTS "Build unit and physics tests" ON)
option(ABMIO_BUILD_BENCHMARKS "Build the benchmark and seed tools" ON)
option(ABMIO_BUILD_FUZZERS "Build fuzz harnesses (libFuzzer with Clang, replay driver otherwise)" OFF)
option(ABMIO_ENABLE_MPI "Build the multi-rank MPI test" OFF)
option(ABMIO_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)
option(ABMIO_COVERAGE "Instrument for coverage" OFF)
set(ABMIO_SANITIZE "" CACHE STRING "Sanitizers, e.g. address,undefined or thread")

# --- LZ4: pkg-config first, plain search as fallback
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
  pkg_check_modules(LZ4 QUIET IMPORTED_TARGET liblz4)
endif()
if(TARGET PkgConfig::LZ4)
  set(ABMIO_LZ4 PkgConfig::LZ4)
else()
  find_path(LZ4_INCLUDE_DIR lz4.h REQUIRED)
  find_library(LZ4_LIBRARY NAMES lz4 REQUIRED)
  add_library(abmio_lz4 INTERFACE)
  target_include_directories(abmio_lz4 INTERFACE ${LZ4_INCLUDE_DIR})
  target_link_libraries(abmio_lz4 INTERFACE ${LZ4_LIBRARY})
  set(ABMIO_LZ4 abmio_lz4)
endif()

# --- shared compile settings
add_library(abmio_flags INTERFACE)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(abmio_flags INTERFACE -Wall -Wextra -Wpedantic -Wshadow -Wnon-virtual-dtor)
  if(ABMIO_WARNINGS_AS_ERRORS)
    target_compile_options(abmio_flags INTERFACE -Werror)
  endif()
  if(ABMIO_SANITIZE)
    target_compile_options(abmio_flags INTERFACE -fsanitize=${ABMIO_SANITIZE} -fno-omit-frame-pointer -fno-sanitize-recover=all)
    target_link_options(abmio_flags INTERFACE -fsanitize=${ABMIO_SANITIZE})
  endif()
  if(ABMIO_COVERAGE)
    target_compile_options(abmio_flags INTERFACE --coverage -O0 -g)
    target_link_options(abmio_flags INTERFACE --coverage)
  endif()
  if(ABMIO_BUILD_FUZZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    target_compile_options(abmio_flags INTERFACE -fsanitize=fuzzer-no-link)
  endif()
endif()

# --- library
add_library(abmio src/exchange.cc src/codec.cc src/generated.cc src/status.cc)
add_library(abmio::abmio ALIAS abmio)
target_include_directories(abmio PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(abmio PUBLIC ${ABMIO_LZ4} PRIVATE abmio_flags)

# --- tests
if(ABMIO_BUILD_TESTS)
  enable_testing()
  find_package(GTest REQUIRED)
  include(GoogleTest)
  add_executable(abmio_tests tests/exchange_test.cc tests/codec_test.cc)
  target_include_directories(abmio_tests PRIVATE tests)
  target_link_libraries(abmio_tests PRIVATE abmio GTest::gtest_main abmio_flags)
  gtest_discover_tests(abmio_tests DISCOVERY_TIMEOUT 120 PROPERTIES LABELS unit)

  add_executable(abmio_physics_tests tests/physics_test.cc)
  target_include_directories(abmio_physics_tests PRIVATE tests)
  target_link_libraries(abmio_physics_tests PRIVATE GTest::gtest_main abmio_flags)
  gtest_discover_tests(abmio_physics_tests DISCOVERY_TIMEOUT 120 PROPERTIES LABELS physics TIMEOUT 600)

  if(ABMIO_ENABLE_MPI)
    find_package(MPI REQUIRED COMPONENTS CXX)
    add_executable(abmio_mpi_test tests/mpi_exchange_test.cc)
    target_include_directories(abmio_mpi_test PRIVATE tests)
    target_link_libraries(abmio_mpi_test PRIVATE abmio MPI::MPI_CXX abmio_flags)
    set(ABMIO_MPI_RANKS 2 CACHE STRING "Ranks for the MPI test")
    add_test(NAME mpi_exchange
             COMMAND ${MPIEXEC_EXECUTABLE} ${MPIEXEC_NUMPROC_FLAG} ${ABMIO_MPI_RANKS} ${MPIEXEC_PREFLAGS}
                     $<TARGET_FILE:abmio_mpi_test> ${MPIEXEC_POSTFLAGS})
    set_tests_properties(mpi_exchange PROPERTIES LABELS mpi SKIP_RETURN_CODE 77 TIMEOUT 300)
  endif()
endif()

# --- benchmarks and seed generator
if(ABMIO_BUILD_BENCHMARKS)
  foreach(tool abmio_bench make_seeds)
    add_executable(${tool} bench/${tool}.cc)
    target_include_directories(${tool} PRIVATE tests)
    target_link_libraries(${tool} PRIVATE abmio abmio_flags)
  endforeach()
endif()

# --- fuzz harnesses
if(ABMIO_BUILD_FUZZERS)
  foreach(f fuzz_unpack fuzz_delta)
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
      add_executable(${f} fuzz/${f}.cc)
      target_link_options(${f} PRIVATE -fsanitize=fuzzer)
    else()
      add_executable(${f} fuzz/${f}.cc fuzz/replay_main.cc)
    endif()
    target_link_libraries(${f} PRIVATE abmio abmio_flags)
  endforeach()
endif()
